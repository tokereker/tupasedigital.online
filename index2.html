function pintarDatos() { document.getElementById('nombre-invitado').innerText = invitadoActual.nombre; let ads = parseInt(invitadoActual.adultos !== undefined ? invitadoActual.adultos : invitadoActual.pases) || 0; let nns = parseInt(invitadoActual.ninos) || 0; document.getElementById('pases-adultos').innerText = ads; document.getElementById('pases-ninos').innerText = nns; // 1. REVISAMOS SI YA RESPONDIÓ if (invitadoActual.status === 'Confirmado' || invitadoActual.status === 'Declinado') { document.getElementById('zona-rsvp').classList.add('hidden'); document.getElementById('zona-estado-actual').classList.remove('hidden'); const textoEstado = document.getElementById('texto-estado'); if (invitadoActual.status === 'Confirmado') { textoEstado.innerText = "¡Asistencia Confirmada! ✅"; textoEstado.className = "text-xl fuente-elegante font-bold text-emerald-600 mb-3"; } else { textoEstado.innerText = "No asistirás ❌"; textoEstado.className = "text-xl fuente-elegante font-bold text-red-500 mb-3"; } } // 2. NUEVA LÓGICA: SI NO HA RESPONDIDO, REVISAMOS LA FECHA LÍMITE else if (baseDeDatosCompleta.config && baseDeDatosCompleta.config.fechaLimite) { const hoy = new Date(); // Convertimos la fecha del CRM (Ej: "2026-10-15") a formato de fecha const limite = new Date(baseDeDatosCompleta.config.fechaLimite); limite.setHours(23, 59, 59); // Le damos hasta las 11:59 PM de ese día if (hoy > limite) { // Si ya pasó la fecha, ocultamos botones y mostramos aviso document.getElementById('zona-rsvp').classList.add('hidden'); document.getElementById('zona-estado-actual').classList.remove('hidden'); const textoEstado = document.getElementById('texto-estado'); textoEstado.innerText = "Tiempo Expirado ⏳"; textoEstado.className = "text-xl fuente-elegante font-bold text-slate-500 mb-3"; // Cambiamos el texto de abajo para que tenga sentido document.querySelector('#zona-estado-actual p.italic').innerText = "El periodo para confirmar asistencia ha finalizado. Por favor, comunícate con el organizador."; } } document.getElementById('pantalla-carga').classList.add('hidden'); document.getElementById('pantalla-invitacion').classList.remove('hidden'); }
